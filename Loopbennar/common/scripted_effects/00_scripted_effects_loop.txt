loop_effelai_hub_lost_effect = {
	limit = {
		has_province_modifier = effelai_hub_attuned
		has_fort_building_trigger = yes
	}
	remove_province_modifier = effelai_tier_1_hub
	remove_province_modifier = effelai_tier_2_hub
	remove_province_modifier = effelai_tier_3_hub
	remove_province_modifier = effelai_tier_4_hub
	remove_province_modifier = effelai_tier_5_hub
	remove_province_modifier = effelai_hub_attuned
	add_permanent_province_modifier = {
		name = effelai_hub_burned
		duration = -1
	}
}
